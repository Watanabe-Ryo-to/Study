①
会員情報を1つずつ所得するSQLを走らせてしまっている

SELECT orders.id,
       orders.member_id,
       orders.order_date,
       orders.total_amount,
       members.name, 
       members.rank,
       members.region
FROM orders

JOIN members ON orders.member_id = members.id

WHERE orders.order_date BETWEEN '2026-05-01' AND '2026-05-31'
  AND orders.status = '発送済';
